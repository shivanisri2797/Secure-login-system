
import sqlite3
import bcrypt
import secrets
import re
import os

from flask import (
    Flask, request, redirect, session,
    render_template_string, g, url_for, abort
)

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY") or secrets.token_hex(32)

DATABASE = "users.db"

# Database connection
def get_db():
    if "db" not in g:
        g.db = sqlite3.connect(DATABASE)
        g.db.row_factory = sqlite3.Row
    return g.db

@app.teardown_appcontext
def close_db(error=None):
    db = g.pop("db", None)
    if db:
        db.close()

# Create database table
def init_db():
    with sqlite3.connect(DATABASE) as db:
        db.execute("""
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                password BLOB NOT NULL
            )
        """)

# Protect forms against CSRF
@app.before_request
def check_csrf():
    if request.method == "POST":
        token = session.get("csrf")
        submitted = request.form.get("csrf", "")
        if not token or not secrets.compare_digest(token, submitted):
            abort(400, "Invalid form token. Reload the page.")

# Home page
@app.route("/")
def home():
    if "csrf" not in session:
        session["csrf"] = secrets.token_urlsafe(32)

    return render_template_string(
        HTML,
        username=session.get("username"),
        message=request.args.get("message", ""),
        csrf=session["csrf"]
    )

# User registration
@app.route("/register", methods=["POST"])
def register():
    username = request.form.get("username", "").strip()
    password = request.form.get("password", "")

    if not re.fullmatch(r"[A-Za-z0-9_]{3,30}", username):
        return redirect(url_for(
            "home", message="Username must contain 3-30 letters, numbers or underscores."
        ))

    if len(password) < 8 or len(password.encode("utf-8")) > 72:
        return redirect(url_for(
            "home", message="Password must be at least 8 characters and at most 72 bytes."
        ))

    hashed_password = bcrypt.hashpw(
        password.encode("utf-8"), bcrypt.gensalt()
    )

    try:
        db = get_db()
        db.execute(
            "INSERT INTO users (username, password) VALUES (?, ?)",
            (username, hashed_password)
        )
        db.commit()
    except sqlite3.IntegrityError:
        return redirect(url_for(
            "home", message="Username already exists."
        ))

    return redirect(url_for(
        "home", message="Registration successful! Please log in."
    ))

# User login
@app.route("/login", methods=["POST"])
def login():
    username = request.form.get("username", "").strip()
    password = request.form.get("password", "")

    user = get_db().execute(
        "SELECT * FROM users WHERE username = ?",
        (username,)
    ).fetchone()

    valid = (
        user is not None
        and len(password.encode("utf-8")) <= 72
        and bcrypt.checkpw(
            password.encode("utf-8"), user["password"]
        )
    )

    if not valid:
        return redirect(url_for(
            "home", message="Invalid username or password."
        ))

    session.clear()
    session["username"] = user["username"]
    session["user_id"] = user["id"]
    session["csrf"] = secrets.token_urlsafe(32)

    return redirect(url_for("home"))

# Logout
@app.route("/logout", methods=["POST"])
def logout():
    session.clear()
    return redirect(url_for(
        "home", message="Logged out successfully."
    ))

if __name__ == "__main__":
    init_db()
    app.run(debug=False)
